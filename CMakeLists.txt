cmake_minimum_required(VERSION 3.22)
project(KernDrum VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
if(JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" JUCE)
else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(KernDrum COMPANY_NAME "Kern Audio" BUNDLE_ID "audio.kern.drums"
  PLUGIN_MANUFACTURER_CODE Krna PLUGIN_CODE Krdm FORMATS VST3 Standalone
  PRODUCT_NAME "KernDrum" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
  IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Instrument Synth)
target_sources(KernDrum PRIVATE Source/Processor.cpp Source/Editor.cpp)
if(MSVC)
  set_source_files_properties(Source/Processor.cpp Source/Editor.cpp PROPERTIES COMPILE_OPTIONS /utf-8)
endif()
target_compile_definitions(KernDrum PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_PLUGINHOST_VST3=1)
target_link_libraries(KernDrum PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
if(MINGW)
  target_link_options(KernDrum PUBLIC -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_executable(CoreTests Tests/CoreTests.cpp)
target_include_directories(CoreTests PRIVATE Source)
add_test(NAME CoreTests COMMAND CoreTests)
juce_add_console_app(IntegrationTests PRODUCT_NAME "KernDrum Integration Tests")
target_sources(IntegrationTests PRIVATE Tests/IntegrationTests.cpp)
target_link_libraries(IntegrationTests PRIVATE KernDrum juce::juce_audio_utils)
add_test(NAME IntegrationTests COMMAND IntegrationTests)
